import json,os
T="/home/it/industrial7_test_crypt/tools/"
mods="crypt32 cryptnet cryptui cryptdlg cryptext cryptxml cryptowinrt cryptcatsvc cryptngc wintrust mssign32 mssip32 capisp tbs keyiso certenc certprop msasn1 ntasn1 certca certcli".split()
def emit(fn,group):
    o=['/* GENERATED from tools/<mod>_exports.json at authoring time (Windows 10 x64 export tables). */','#include "exp.h"']
    for m in group:
        d=sorted(json.load(open(T+m+"_exports.json")),key=lambda x:x['ordinal'])
        o.append("const exp_ent exp_tab_%s[]={"%m)
        for x in d:
            n='"%s"'%x['name'] if x['name'] else 'NULL'
            f='"%s"'%x['forwarder'] if x['forwarder'] else 'NULL'
            o.append(' {%s,%d,%s},'%(n,x['ordinal'],f))
        o.append("};")
        o.append("const int exp_cnt_%s=%d;"%(m,len(d)))
    open(fn,"w").write("\n".join(o)+"\n")
